{
    //debugger;
    return conn.query("SELECT ProductCode__c, DeltaCalcOnAmendment__c FROM CustomScript_Config__mdt WHERE DeltaCalcOnAmendment__c = true")
        .then(function(result) {
            cpqProductConfig = { deltaCalc: new Set() };
            (result.records || []).forEach(function(r) {
                cpqProductConfig.deltaCalc.add(r.ProductCode__c);
            });
            populateGetSeasonalPriceTable(quote,lines);
            copyQuantityOnClonedlines(quote, lines);

            }

/*
Process Promo Product Consumption Rates
Method - processPromoProductsCSRates
called via onAfterPriceRules
*/
export function processPromoProductsCSRates( lines , promoProducts , isSeasonal) {

    if( lines ){
        lines.forEach(function(line){
            if ( line && line.record ){
                let percentDiscount = line.record.SBQQ__Discount__c;
                let amountDiscount = line.record.SBQQ__AdditionalDiscountAmount__c;
                let productcode = line.record.SBQQ__ProductCode__c;
                let linequantity = line.record.NSP_Quoted_Quantity__c;
                let netUnitPrice =  line.record["SBQQ__NetPrice__c"];
                let discountType = "";
                if(percentDiscount === null || percentDiscount === ""){
                    if(amountDiscount === null || amountDiscount === ""){
                        discountType = "no discount";
                    }else{
                        discountType = "amount";
                    }
                }
                if(line.consumptionSchedules){
                    line.consumptionSchedules.forEach(function(cs){
                        if(productcode && promoProducts && promoProducts.includes(productcode)){
                            const rates = cs.getRates();
                            rates.forEach(function(csRate){
                                let upperbond = csRate.get('SBQQ__UpperBound__c');
                                let lowerbond = csRate.get('SBQQ__LowerBound__c');
                                if(isSeasonal && lowerbond <= linequantity && upperbond > linequantity){
                                    csRate.set('SBQQ__Price__c', netUnitPrice);
                                }
                                let currentPrice =  csRate.get("SBQQ__Price__c");
                                let pricingMethod = csRate.get("SBQQ__PricingMethod__c");
                                if( pricingMethod != 'FlatFee' ){
                                    let newPriceAmt = 0;
                                    switch (discountType){
                                        case "no discount":
                                            newPriceAmt = -1 * currentPrice;
                                            csRate.set("SBQQ__Price__c", newPriceAmt);
                                            newPriceAmt = 0;
                                            break;
                                        case "amount":
                                            newPriceAmt = -1 * currentPrice;
                                            csRate.set("SBQQ__Price__c", newPriceAmt);
                                            newPriceAmt = 0;
                                            break;
                                    }
                                }
                            });
                        }
                    });
                }
            }
        });
    }
}

/*
Normal Pricing Implementation
Method - ProcessRates
called via onBeforeCalculate
US - 299910,294434
*/
export function ProcessRates(quote,lines,csProducts,csRates){

    if(quote.record.SBQQ__LineItemCount__c>0){
        lines.forEach(function(line){
            let productcode = line.record.SBQQ__ProductCode__c;
            if(line.consumptionSchedules){
                line.consumptionSchedules.forEach(function(cs){
                    let CSName = cs.get("Name");
                    if(csRates.includes(CSName) && csProducts.includes(productcode)){
                        const rates = cs.getRates();
                        rates.forEach(function(csRate){
                            if(csRate){
                                let descriptionField = csRate.get("SBQQ__Description__c");
                                if(descriptionField){
                                    let splitArray = descriptionField.split("_");
                                    if(splitArray.length > 1){
                                        let priceField = csRate.get("SBQQ__Price__c");
                                        let originalPriceField = csRate.get("SBQQ__OriginalPrice__c");

                                        if(priceField !== parseFloat(splitArray[0])){
                                            priceField = parseFloat(splitArray[0]);
                                            csRate.set("SBQQ__Price__c", priceField);
                                        }
                                        if(originalPriceField !== parseFloat(splitArray[1])){
                                            originalPriceField = parseFloat(splitArray[1]);
                                            csRate.set("SBQQ__OriginalPrice__c", originalPriceField);
                                        }
                                    }
                                }
                            }
                            let currentPrice = csRate.get("SBQQ__Price__c");
                            let originalPrice = csRate.get("SBQQ__OriginalPrice__c");
                            let descriptionArray = [];
                            let newSetPrice = currentPrice;

                            if(originalPrice !== currentPrice){
                                csRate.set("SBQQ__Price__c", originalPrice);
                            }
                            let newPrice = originalPrice;
                            csRate.set("SBQQ__Price__c", newPrice);
                            let newDescription = newPrice + "_" + originalPrice;
                            csRate.set("SBQQ__Description__c", newDescription);
                        });
                    }
                });
            }
        });
    }
}

/*
Consumption Schedule Discount Implementation
Method - processCSDiscount
called via onAfterPriceRules
US - 299910,294434
*/
export function processCSDiscount(quote,lines,csProducts,csRates){
    let quotediscountCS = quote.record.Discount_Consumption_Schedules__c;

    //get the discount type(percent deducted as %, Dollar deducted directly)
    lines.forEach(function(line){
        let percentDiscount = line.record.SBQQ__Discount__c;
        let amountDiscount = line.record.SBQQ__AdditionalDiscountAmount__c;
        let finalMinFee = line.record.NSP_Final_Minimum_Fee__c;
        let productcode = line.record.SBQQ__ProductCode__c;
        let lineDiscount = 0;
        let discountType = "";
        if(percentDiscount === null || percentDiscount === ""){
            if(amountDiscount === null || amountDiscount === ""){
                lineDiscount = 0;
                discountType = "no discount";
            }else{
                lineDiscount = amountDiscount;
                discountType = "amount";
            }
        }else{
            lineDiscount = percentDiscount;
            discountType = "percent";
        }

        if(line.consumptionSchedules){
            line.consumptionSchedules.forEach(function(cs){
                let CSName = cs.get("Name");

                if(csRates.includes(CSName) && csProducts.includes(productcode)){
                    const rates = cs.getRates();
                    rates.forEach(function(csRate){
                        let originalPrice = csRate.get("SBQQ__OriginalPrice__c");
                        let currentPrice =  csRate.get("SBQQ__Price__c");
                        let pricingMethod = csRate.get("SBQQ__PricingMethod__c");
                        //newSetPrice is used for calculation purpose to be populated below
                        let newSetPrice = 0;
                        //Description stores the non discounted rolled up price
                        let descriptionField = csRate.get("SBQQ__Description__c");
                        if(descriptionField && descriptionField.length>0){
                            let splitArray = descriptionField.split("_");
                            if(splitArray.length > 1){
                                //use the non discounted rolled up price of the rate
                                newSetPrice = parseFloat(splitArray[0]);
                            }else{
                                //SBQQ__Price__c is already the non discounted rolled up price
                                newSetPrice = currentPrice;
                            }
                            if(newSetPrice!==currentPrice){
                                csRate.set("SBQQ__Price__c", newSetPrice);
                                currentPrice = newSetPrice;
                            }
                        }
                        if(csRates.includes(CSName) && csProducts.includes(productcode) && pricingMethod != 'FlatFee'){
                            let newPriceAmt = 0;
                            let newPricePer = 0;
                            switch (discountType){
                                case "no discount":
                                    break;
                                case "amount":
                                    newPriceAmt = currentPrice - lineDiscount;
                                    csRate.set("SBQQ__Price__c", newPriceAmt);
                                    newPriceAmt = 0;
                                    originalPrice = 0;
                                    break;
                                case "percent":
                                    newPricePer = currentPrice * ((100-lineDiscount)/100);
                                    csRate.set("SBQQ__Price__c", newPricePer);
                                    newPricePer = 0;
                                    originalPrice = 0;
                                    break;
                            }
                        }
                        else if(csRates.includes(CSName) && csProducts.includes(productcode) && pricingMethod == 'FlatFee' && finalMinFee != null)
                        {
                            csRate.set("SBQQ__Price__c", finalMinFee);
                        }else{
                           if(originalPrice !== currentPrice && originalPrice != null){
                                csRate.set("SBQQ__Price__c", originalPrice);
                            }
                        }
                    });
                }
            });
        }
    });
}

export function undoQuantityUpdateonTerm(quoteModel, quoteLineModels) {
    debugger;
    let terminateAll =  quoteModel.record["NSP_Terminate_All__c"];
    let quoteType =  quoteModel.record["SBQQ__Type__c"];
    let quoteTransType = quoteModel.record["NSP_TransactionType__c"];

    if(quoteType == 'Amendment' && quoteTransType == 'Termination' && terminateAll == false)
            {
                quoteLineModels.forEach(function(line) {
                if(line.record["NSP_Previous_Quantity__c"] == null) {
                    line.record["NSP_Previous_Quantity__c"] = line.record["NSP_Quoted_Quantity__c"];
                }
                if(line.record["NSP_Term_Line__c"] == true && line.record["NSP_Terminate_All__c"] == true) {
                    if(line.record["NSP_Lock_On_QLI__c"] == true) {
                        line.record["SBQQ__Quantity__c"] = 1;
                        line.record["NSP_Quoted_Quantity__c"] = 1;
                        line.record["NSP_Term_Line__c"] = false;
                        line.record["NSP_Terminate_All__c"] = false;
                    } else if((line.record["NSP_Lock_On_QLI__c"] == false) && line.record["SBQQ__SubscriptionType__c"] == 'Evergreen') {
                        let upgradedSubs = line.RequiredBy__r;
                        line.record["SBQQ__Quantity__c"] = 1;
                        line.record["NSP_Term_Line__c"] = false;
                        line.record["NSP_Terminate_All__c"] = false;
                        line.record["NSP_Quoted_Quantity__c"] = line.record["NSP_Previous_Quantity__c"];
                    } else {
                        line.record["SBQQ__Quantity__c"] = line.record["SBQQ__PriorQuantity__c"];
                        line.record["NSP_Quoted_Quantity__c"] = line.record["SBQQ__PriorQuantity__c"];
                        line.record["NSP_Term_Line__c"] = false;
                        line.record["NSP_Terminate_All__c"] = false;
                    }
                } else if(line.record["NSP_Term_Line__c"] == true && line.record["NSP_Terminate_All__c"] == false && line.record["NSP_Lock_On_QLI__c"] == true) {
                    line.record["SBQQ__Quantity__c"] = 0;
                    line.record["NSP_Quoted_Quantity__c"] = 0;
                } else if(line.record["NSP_Term_Line__c"] == false && line.record["NSP_Terminate_All__c"] == false && line.record["NSP_Lock_On_QLI__c"] == true) {
                    line.record["SBQQ__Quantity__c"] = line.record["SBQQ__PriorQuantity__c"];
                    line.record["NSP_Quoted_Quantity__c"] = line.record["SBQQ__PriorQuantity__c"];;
                }


            });
            }
}

export function copyQuantityOnClonedlines(quote, lines){
    debugger;
    var productcountMap =  new Map();
    lines.forEach(function(line){
        let productId = line.record["SBQQ__Product__c"];
        let subscriptiontype = line.record["SBQQ__SubscriptionType__c"];
        if(productId && subscriptiontype){
            productcountMap.set(productId+'_'+subscriptiontype,line);
        }
    });

    lines.forEach(function(line){
        let productId = line.record["SBQQ__Product__c"];
        let isclonedline = line.record["NSP_IsClonedLine__c"];
        if(productId && isclonedline && productcountMap.has(productId+'_'+'Renewable') ) {
            line.record["SBQQ__Quantity__c"] = productcountMap.get(productId+'_'+'Renewable').record["SBQQ__Quantity__c"];
        }
    });
}

            if ( ( productId == 'IWXSU' )

                && terminatingWoWs
                && !stopAutoPricing

            {
                line.record["SBQQ__ListPrice__c"] = 0;
            }
        }
    );
}

/*
Seasonal Pricing Implementation
Method - processSeasonalDiscount
called via onAfterPriceRules
US - 299910,294434
*/
export function processSeasonalDiscount(quote,lines){
    let seasonalQuote = quote.record.Seasonal_Pricing__c;

    //get the discount type(percent deducted as %, Dollar deducted directly)
    lines.forEach(function(line){
        let percentDiscount = line.record.SBQQ__Discount__c;
        let amountDiscount = line.record.SBQQ__AdditionalDiscountAmount__c;
        let lineDiscount = 0;
        let discountType = "";
        if(percentDiscount === null || percentDiscount === ""){
            if(amountDiscount === null || amountDiscount === ""){
                lineDiscount = 0;
                discountType = "no discount";
            }else{
                lineDiscount = amountDiscount;
                discountType = "amount";
            }
        }else{
            lineDiscount = percentDiscount;
            discountType = "percent";
        }

        if(line.consumptionSchedules){
            line.consumptionSchedules.forEach(function(cs){
                /*add condition on cs for checking seasonal*/
                let isSeasonal = cs.get("Name");
                if(isSeasonal === seasonalQuote){
                    const rates = cs.getRates();
                    processSeasonalDiscountHelper(rates
                        , discountType
                        , lineDiscount
                        , isSeasonal
                        , seasonalQuote
                    );
                }
            });
        }
    });
}

/*
Seasonal Pricing Implementation
Method - processSeasonalDiscountHelper
called via processSeasonalDiscount
US - 299910,294434
*/
export function processSeasonalDiscountHelper(
    rates
    , discountType
    , lineDiscount
    , isSeasonal
    , seasonalQuote
){
    rates.forEach(function(csRate){
        let originalPrice = csRate.get("SBQQ__OriginalPrice__c");
        let currentPrice =  csRate.get("SBQQ__Price__c");
        //newSetPrice is used for calculation purpose to be populated below
        let newSetPrice = 0;
        //Description stores the non discounted rolled up price
        let descriptionField = csRate.get("SBQQ__Description__c");
        if(descriptionField){
            let splitArray = descriptionField.split("_");
            if(splitArray.length > 1){
                //use the non discounted rolled up price of the rate
                newSetPrice = parseFloat(splitArray[1]);
            }else{
                //SBQQ__Price__c is already the non discounted rolled up price
                newSetPrice = currentPrice;
            }
        }
        if(newSetPrice!==currentPrice){
            csRate.set("SBQQ__Price__c", newSetPrice);
            currentPrice = newSetPrice;
        }
        if(isSeasonal === seasonalQuote){
            let newPriceAmt = 0;
            let newPricePer = 0;
            switch (discountType){
                case "no discount":
                    break;
                case "amount":
                    newPriceAmt = currentPrice - lineDiscount;
                    csRate.set("SBQQ__Price__c", newPriceAmt);
                    newPriceAmt = 0;
                    originalPrice = 0;
                    break;
                case "percent":
                    newPricePer = currentPrice * ((100-lineDiscount)/100);
                    csRate.set("SBQQ__Price__c", newPricePer);
                    newPricePer = 0;
                    originalPrice = 0;
                    break;
            }
        }else{
            if(originalPrice !== currentPrice){
                csRate.set("SBQQ__Price__c", originalPrice);
            }
        }
    });
}

/*
Seasonal Pricing Implementation
Method - ProcessSeasonalRates
called via onBeforeCalculate
US - 299910,294434
*/
export function ProcessSeasonalRates(quote,lines){
    let seasonalQuote = quote.record.Seasonal_Pricing__c;

    if(quote.record.SBQQ__LineItemCount__c>0){
        //Map to store Bundle parent id as key-
        //-and Seasonal products in bundle as array
        let seasonalMap = new Map();
        //iterate through all lines to gather product codes
        lines.forEach(function(line){
            let productCode = line.record.SBQQ__ProductCode__c;
            let SeasonalApplicable = line.record.Seasonal_Pricing__c;
            let requiredby = line.record.SBQQ__RequiredBy__c;
            if((SeasonalApplicable !== "")
                 && (SeasonalApplicable !== "Not Applicable")
                 && (requiredby !== "")
                 && (requiredby !== null)
                ){
                if(seasonalMap.has(requiredby)){
                    seasonalMap.get(requiredby).push(productCode);
                }else{
                    seasonalMap.set(requiredby,[productCode]);
                }
            }
        });
        //now we have all the product codes on which seasonal is applicable
        lines.forEach(function(line){
            let lineKey = line.record.Id;
            if(line.consumptionSchedules){
                line.consumptionSchedules.forEach(function(cs){
                    let isSeasonal = cs.get("Name");
                    if(isSeasonal === seasonalQuote){
                        const rates = cs.getRates();
                        rates.forEach(function(csRate){
                            ProcessSeasonalRatesHelper(csRate
                                , seasonalMap
                                , lineKey
                            );
                        });
                    }
                });
            }
        });
    }
}

/*
Seasonal Pricing Implementation
Method - ProcessSeasonalRatesHelper
called via ProcessSeasonalRates
US - 299910,294434
*/
export function ProcessSeasonalRatesHelper(csRate
    , seasonalMap
    , lineKey
){
    resetSeasonalRate(csRate);
    let currentPrice = csRate.get("SBQQ__Price__c");
    let originalPrice = csRate.get("SBQQ__OriginalPrice__c");
    let descriptionArray = [];
    let newSetPrice = currentPrice;

    if(originalPrice !== currentPrice){
        csRate.set("SBQQ__Price__c", originalPrice);
    }
    let newPrice = originalPrice;
    let descriptionField = csRate.get("SBQQ__Description__c");
    if(descriptionField){
        let splitArray = descriptionField.split("_");
        if(splitArray.length > 1){
            descriptionArray = splitArray[0];
            newSetPrice = parseFloat(splitArray[1]);
        }
        else{
            descriptionArray = descriptionField;
        }
    }
    if(descriptionArray){
        const productPrices = JSON.parse(descriptionArray);
        const productEnteriesObject = Object.entries(productPrices);
        productEnteriesObject.forEach(function([key, value]){
            if(seasonalMap.has(lineKey)){
                let productCodeListFromMap = seasonalMap.get(lineKey);
                if(productCodeListFromMap.includes(key)){
                    newPrice += value;
                }
            }
        });
        csRate.set("SBQQ__Price__c", newPrice);
        let newDescription = descriptionArray + "_" + newPrice + "_" + originalPrice;
        csRate.set("SBQQ__Description__c", newDescription);
    }
}

/*
Seasonal Pricing Implementation
Method - resetSeasonalRate
called via ProcessSeasonalRatesHelper
US - 299910,294434
*/
export function resetSeasonalRate(csRate){
    //method used to reset the original price
    //to prevent CPQ from overiding the OG price
    if(csRate){
        let descriptionField = csRate.get("SBQQ__Description__c");
        if(descriptionField){
            let splitArray = descriptionField.split("_");
            if(splitArray.length > 1){
                let priceField = csRate.get("SBQQ__Price__c");
                let originalPriceField = csRate.get("SBQQ__OriginalPrice__c");
                if(priceField !== parseFloat(splitArray[1])){
                    priceField = parseFloat(splitArray[1]);
                    csRate.set("SBQQ__Price__c", priceField);
                }if(originalPriceField !== parseFloat(splitArray[2])){
                    originalPriceField = parseFloat(splitArray[2]);
                    csRate.set("SBQQ__OriginalPrice__c", originalPriceField);
                }
            }
        }
    }
}

                let scholarshipApplicable = line.record["NSP_Scholarship_Price__c"];
                let listPrice = line.record["SBQQ__ListPrice__c"];
                let originalMinFee = line.record["NSP_Original_List_Minimum_Fee__c"];
                let linequantity = line.record["NSP_Quoted_Quantity__c"];
                let maxquantity = line.record["NSP_Max_Quantity__c"];
                let productcode = line.record["SBQQ__ProductCode__c"];
                let thresholdapplicable = line.record["NSP_Threshold_Applicable__c"];
                let totalDiscountAmount = line.record["SBQQ__TotalDiscountAmount__c"];
                let addDiscPercent = line.record["SBQQ__Discount__c"];
                let flatFeeReferencePrice = line.record["Flat_Fee_Reference_Price__c"];
                let flatfeeUpperbondvalue;
                let flatfeeprice = finalMinFee;
                let unitPriceThresholduantity;
                let nPerUnitContWorkforce = 0;

                if(maxquantity && linequantity && linequantity <= maxquantity && finalMinFee >= 0){
                    if(finalMinFee == 0 && thresholdPercentage >= 0) {
                        flatfeeUpperbondvalue = 1;
                        flatfeeprice = 0;
                    } else {
                        flatfeeUpperbondvalue = parseInt(finalMinFee/netUnitPrice)+1;
                    }
                }
                else if(thresholdapplicable && thresholdPercentage && linequantity){
                    flatfeeUpperbondvalue = parseInt(0.01*thresholdPercentage * linequantity)+1;
                    flatfeeprice = (0.01*thresholdPercentage * linequantity) * netUnitPrice;
                    unitPriceThresholduantity = (0.01*thresholdPercentage * linequantity) * listPrice;
                }
                else if(contractorProductlist.includes(productcode) && maxquantity && linequantity && linequantity > maxquantity && finalMinFee >= 0){
                    flatfeeUpperbondvalue = parseInt(finalMinFee/netUnitPrice)+1;
                }
                else if(maxquantity && linequantity && linequantity > maxquantity && finalMinFee>=0){
                    if(finalMinFee == 0 && thresholdPercentage >= 0) {
                        flatfeeUpperbondvalue = 1;
                        flatfeeprice = 0;
                    }else{
                        flatfeeUpperbondvalue = parseInt(finalMinFee/netUnitPrice)+1;
                    }

                }
                if(maxquantity && linequantity && linequantity > maxquantity && thresholdPercentage == 0 && thresholdapplicable){
                    flatfeeUpperbondvalue = 1;
                        flatfeeprice = 0;
                }

                if(applicableProductlist.includes(productcode) && productcode != 'ITSUserLicenseWO'  ){
                        quote.record["NSP_WX_Upper_Bound__c"] = flatfeeUpperbondvalue ? flatfeeUpperbondvalue : null;
                        line.record["NSP_UsageUpperBound__c"] = flatfeeUpperbondvalue ? flatfeeUpperbondvalue - 1 : null;
                    }

                if (line.consumptionSchedules) {

                    line.consumptionSchedules.forEach(function(cs){
                        const rates = cs.getRates();
                        let ratecount = rates.length;
                        let matchingattribute = cs.get('SBQQ__MatchingAttribute__c');
                        let Name = cs.get("Name");
                        if(String(Name).substring(0,8) === "Seasonal"){
                            return;
                        }
                        if(!matchingattribute || !matchingattribute.toLowerCase().includes('partner')){

                            rates.forEach(function(csrate,index){
                                let upperbond = csrate.get('SBQQ__UpperBound__c');
                                let lowerbond = csrate.get('SBQQ__LowerBound__c');
                                let processingorder = csrate.get('SBQQ__ProcessingOrder__c');
                                let pricingMethod = csrate.get('SBQQ__PricingMethod__c');


                                debugger;
                                if(!minmumfeeVal && productcode && !applicableProductlist.includes(productcode) && slabProducts && !slabProducts.includes(productcode))// for product with non min fee
                                {
                                    if(lowerbond <= linequantity && upperbond > linequantity)
                                    {
                                        csrate.set('SBQQ__Price__c', netUnitPrice);
                                    }
                                }
                                else if(productcode && applicableProductlist.includes(productcode) && !scholarshipApplicable )
                                { // scholarship not applicable

                                    if( pricingMethod == 'FlatFee' && flatfeeprice >= 0 && flatfeeUpperbondvalue)
                                    {
                                        csrate.set('SBQQ__Price__c', flatfeeprice);
                                        csrate.set('SBQQ__UpperBound__c', flatfeeUpperbondvalue);

                                    }
                                    else if(pricingMethod == 'PerUnit' && flatfeeUpperbondvalue){
                                        csrate.set('SBQQ__Price__c', netUnitPrice);
                                        csrate.set('SBQQ__LowerBound__c', flatfeeUpperbondvalue);
                                    }

                                    if(!wxscholarship && matchingattribute == 'Scholarship'){ // update price of scholarschip cs ro zero
                                            csrate.set('SBQQ__Price__c', 0); // setting scholarship CS price to zero when Wx scholarhip is false STY-01818
                                    }

                                }
                                else if(productcode && applicableProductlist.includes(productcode) && scholarshipApplicable )
                                { // scholarship  applicable

                                    let flatfeeval = flatfeeprice;
                                    let perunitval = netUnitPrice;
                                    let nonSchFlatfee = unitPriceThresholduantity;

                                        if(maxquantity && linequantity && linequantity <= maxquantity && finalMinFee)
                                        {
                                            //flatfeeval = minmumfeeVal;
                                            perunitval = listPrice;
                                            nonSchFlatfee = minmumfeeVal;
                                        }

                                        if(!matchingattribute)
                                        {
                                            if( pricingMethod == 'FlatFee' && flatfeeprice >= 0 && flatfeeUpperbondvalue   ) // update non scholarship cs
                                            {
                                                csrate.set('SBQQ__Price__c', nonSchFlatfee);
                                                    csrate.set('SBQQ__UpperBound__c', parseInt(nonSchFlatfee/listPrice)+1);

                                            }
                                            else if(pricingMethod == 'PerUnit' && flatfeeUpperbondvalue){
                                                csrate.set('SBQQ__Price__c', listPrice);
                                                    csrate.set('SBQQ__LowerBound__c', parseInt(nonSchFlatfee/listPrice)+1);
                                            }
                                        }
                                        else if(matchingattribute == 'Scholarship')
                                        {
                                            if( pricingMethod == 'FlatFee' && flatfeeprice >= 0 && flatfeeUpperbondvalue  ) // update scholarship cs
                                            {
                                                csrate.set('SBQQ__Price__c', flatfeeval);
                                                    csrate.set('SBQQ__UpperBound__c', parseInt(flatfeeval/netUnitPrice)+1);

                                            }
                                            else if(pricingMethod == 'PerUnit' && flatfeeUpperbondvalue){
                                                csrate.set('SBQQ__Price__c', netUnitPrice);
                                                    csrate.set('SBQQ__LowerBound__c', parseInt(flatfeeval/netUnitPrice)+1);
                                            }
                                        }
                                }
                                else if(productcode && contractorProductlist.includes(productcode)){ // for contractors

                                    if( pricingMethod == 'FlatFee' && finalMinFee >= 0 && flatfeeUpperbondvalue)
                                    {
                                        csrate.set('SBQQ__Price__c', finalMinFee);
                                        csrate.set('SBQQ__UpperBound__c', flatfeeUpperbondvalue);
                                    }
                                    else if(pricingMethod == 'PerUnit' && flatfeeUpperbondvalue){
                                        csrate.set('SBQQ__Price__c', netUnitPrice);
                                        csrate.set('SBQQ__LowerBound__c', flatfeeUpperbondvalue);
                                    }
                                }
                                else if( flatFeeReferencePrice && customProductlist.includes(productcode) && linequantity > 0){ // for contractors

                                    if( pricingMethod == 'FlatFee' && flatFeeReferencePrice >= 0 && flatfeeUpperbondvalue && finalMinFee !=null)
                                    {
                                        csrate.set('SBQQ__Price__c', finalMinFee);
                                        if(addDiscPercent)
                                        {
                                            csrate.set('SBQQ__UpperBound__c', parseInt(finalMinFee/(flatFeeReferencePrice-(addDiscPercent*0.01*flatFeeReferencePrice))) +1);
                                        }
                                        else
                                        {
                                            csrate.set('SBQQ__UpperBound__c', parseInt(finalMinFee/(flatFeeReferencePrice-totalDiscountAmount)) +1);
                                        }
                                    }
                                    else if(pricingMethod == 'PerUnit' && flatFeeReferencePrice >=0 && nPerUnitContWorkforce <= 0 && netUnitPrice !=null){
                                        csrate.set('SBQQ__Price__c', netUnitPrice);
                                        if(addDiscPercent)
                                        {
                                            csrate.set('SBQQ__LowerBound__c', parseInt(finalMinFee/(flatFeeReferencePrice-(addDiscPercent*0.01*flatFeeReferencePrice))) +1);
                                        }
                                        else
                                        {
                                            csrate.set('SBQQ__LowerBound__c', parseInt(finalMinFee/(flatFeeReferencePrice-totalDiscountAmount))+1);
                                        }
                                        nPerUnitContWorkforce = nPerUnitContWorkforce +1;
                                    }
                                }
                                else{
                                    if(slabProducts && !slabProducts.includes(productcode)){
                                        csrate.set('SBQQ__Price__c', netUnitPrice);
                                    }
                                }
                            });
                        }
                    });
                }
            });
        }
    }
}

export function amendmentForClonedLineCalculation (quote, lines){
    let quoteType = quote.record["SBQQ__Type__c"];
    let quoteTransType = quote.record["NSP_TransactionType__c"];
    let productMap = new Map();
    if(quoteType == 'Amendment' && quoteTransType != 'Termination'){
        lines.forEach(function(line){
            if(line.record["SBQQ__UpgradedSubscription__c"]!=null && line.record["SBQQ__Bundle__c"]== false && (line.record["SBQQ__PricingMethod__c"]=='Block' || cpqProductConfig.deltaCalc.has(line.record["SBQQ__ProductCode__c"]))){
                //getting source line's list price
                productMap.set(line.record["SBQQ__ProductCode__c"],line.record["SBQQ__ListPrice__c"]);
            }
        });

        lines.forEach(function(line){

            if(productMap.has(line.record["SBQQ__ProductCode__c"]) && line.record["SBQQ__Source__c"]!=null){
                let isDeltaCalc = cpqProductConfig.deltaCalc.has(line.record["SBQQ__ProductCode__c"]);
                let deltaPrice;
                if(isDeltaCalc){
                    let effectiveQty = line.record["SBQQ__EffectiveQuantity__c"] || 0;
                    let prevQty = line.record["NSP_Previous_Quantity__c"] || 0;
                    let quotedQty = line.record["NSP_Quoted_Quantity__c"] || 0;
                    let deltaQty = quotedQty - prevQty;
                    deltaPrice = effectiveQty > 0 ? deltaQty * line.record["SBQQ__ListPrice__c"] : 0;
                } else {
                    deltaPrice = productMap.get(line.record["SBQQ__ProductCode__c"])<0?line.record["SBQQ__ListPrice__c"]+productMap.get(line.record["SBQQ__ProductCode__c"]):line.record["SBQQ__ListPrice__c"]-productMap.get(line.record["SBQQ__ProductCode__c"]);
                }
                line.record["NSP_Delta_Total__c"] = deltaPrice;
                let disc = line.record["NSP_Delta_Discount__c"];
                let discountedValue  = deltaPrice-[disc];

                if(line.record["SBQQ__DefaultSubscriptionTerm__c"]==1 && line.record["SBQQ__BillingFrequency__c"] !='Annual'){
                    let netPrice  = discountedValue + line.record["NSP_Original_Price__c"];
                    line.record["SBQQ__CustomerPrice__c"] = netPrice;
                    line.record["SBQQ__NetPrice__c"] = netPrice;
                    line.record["NSP_Delta_Total__c"] = line.record["NSP_Delta_Total__c"] - disc;

                }
                if(line.record["SBQQ__DefaultSubscriptionTerm__c"]>1 && line.record["SBQQ__BillingFrequency__c"] !='Annual'){
                    let subTerm = line.record["SBQQ__SubscriptionTerm__c"]!=null?line.record["SBQQ__SubscriptionTerm__c"]:1;
                    let netPrice  = discountedValue*subTerm/12 + line.record["NSP_Original_Price__c"];
                    line.record["SBQQ__CustomerPrice__c"] = netPrice;
                    line.record["SBQQ__NetPrice__c"] = netPrice;
                    line.record["NSP_Delta_Total__c"] = line.record["NSP_Delta_Total__c"] - disc;
                }
                if(line.record["SBQQ__BillingFrequency__c"] == 'Annual'){
                    line.record["NSP_Delta_Total__c"] = line.record["NSP_Delta_Total__c"] - disc;
                    line.record["SBQQ__NetPrice__c"] = line.record["SBQQ__NetPrice__c"] - disc;
                }
            }

        });
    }
}

export function checkSerialIdValidationFlag(quote, lines){
    let quoteType = quote.record["SBQQ__Type__c"];
    let quoteTransType = quote.record["NSP_TransactionType__c"];
    if((quoteType=='Amendment' && quoteTransType != 'Termination') ||quoteType=='Renewal' ){
        lines.forEach(function(line){
            let actionType=line.record["NSP_Action_Type__c"];
            let actionTypeF=line.record["NSP_ActionType__c"];
            let createEntitlement=line.record["NSP_Create_Entitlement__c"];
            let quotedQty=line.record["NSP_Quoted_Quantity__c"];
            let serialIds=line.record["NSP_Serial_Ids__c"];
            let prevQty=line.record["NSP_Previous_Quantity__c"];
            let prevSubQty=line.record["NSP_Previous_subQty__c"];
            line.record["NSP_SerialId_Validation__c"]=false; //clearing the value initially

            if(actionType=='Renewal' && createEntitlement===true){
                let diffQty=Math.abs(prevSubQty-quotedQty);
                if((serialIds=='' ||serialIds==undefined||serialIds==null) && diffQty>0){
                    line.record["NSP_SerialId_Validation__c"]=true;
                }
                if(serialIds!='' && serialIds){
                    let serialArray=serialIds.split(",").length;
                    line.record["NSP_SerialId_Validation__c"]=serialArray!=diffQty?true:false;
                }
            }
            if(createEntitlement==true && (actionTypeF=='Downgrade'||actionTypeF=='Upsell')){
                let diffQty1=Math.abs(quotedQty-prevQty);
                if(serialIds=='' ||serialIds==undefined ||serialIds==null){
                    line.record["NSP_SerialId_Validation__c"]=true;
                }
                if(serialIds!='' && serialIds){
                    let serialArray1=serialIds.split(",").length;
                    line.record["NSP_SerialId_Validation__c"]=serialArray1!=diffQty1?true:false;
                }
            }

        })
    }
